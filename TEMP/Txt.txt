class _HomeScreenState extends State<HomeScreen> {
  late Future<List<Medication>> _medsFuture;

  @override
  void initState() {
    super.initState();
    _medsFuture = MedRepository().getAllMeds();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('MedAdhere'),
        backgroundColor: Colors.green,
      ),
      body: FutureBuilder<List<Medication>>(
        future: _medsFuture,
        builder: (context, snapshot) {
          if (snapshot.connectionState == ConnectionState.waiting) {
            return const Center(child: CircularProgressIndicator());
          }
          if (snapshot.hasError) {
            return Center(child: Text('Error: ${snapshot.error}'));
          }
          final meds = snapshot.data ?? [];
          if (meds.isEmpty) {
            return const Center(
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  Icon(Icons.poll, size: 64, color: Colors.grey),
                  SizedBox(height: 16),
                  Text(
                    'No medications yet',
                    style: TextStyle(color: Colors.grey),
                  ),
                ],
              ),
            );
          }
          return ListView.builder(
            padding: const EdgeInsets.all(16),
            itemCount: meds.length,
            itemBuilder: (context, index) {
              final med = meds[index];
              final formattedTimes = med.times.map(_formatTime12).join(', ');
              return Dismissible(
                key: Key(med.id), // Required for Dismissible
                direction: DismissDirection.endToStart,
                onDismissed: (direction) async {
                  // Delete from Hive
                  await med.delete(); // HiveObject has built-in .delete()

                  // Remove from local list (for immediate UI update)
                  setState(() {
                    meds.removeAt(index);
                  });

                  // Show confirmation
                  ScaffoldMessenger.of(context).showSnackBar(
                    SnackBar(
                      content: Text('${med.name} removed'),
                      action: SnackBarAction(
                        label: 'Undo',
                        onPressed: () {
                          // Optional: re-add (advanced — skip for MVP)
                        },
                      ),
                    ),
                  );
                },
                background: Container(
                  color: Colors.red,
                  alignment: Alignment.centerRight,
                  padding: const EdgeInsets.only(right: 20),
                  child: const Icon(Icons.delete, color: Colors.white),
                ),
                child: Card(
                  child: ListTile(
                    title: Text(med.name),
                    subtitle: Text(
                      '${med.dosage} • $formattedTimes • ${_formatFrequency(med.frequency)}',
                    ),
                    trailing: const Icon(Icons.arrow_forward_ios, size: 16),
                  ),
                ),
              );
            },
          );
        },
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: () => context.go('/add-med'),
        backgroundColor: Colors.green,
        child: const Icon(Icons.add, color: Colors.white),
      ),
    );
  }
}
